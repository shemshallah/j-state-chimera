def investigate_circuit():
    """Check if circuit creates the intended Janus state"""
    
    print("\n" + "="*80)
    print("LEVEL 1: CIRCUIT VERIFICATION")
    print("="*80)
    
    print("\n[1.1] Build Janus state circuit")
    qc = QuantumCircuit(4)
    
    # Your circuit definition
    qc.h(0)
    qc.ch(0, 1)
    qc.cx(0, 2)
    qc.cx(1, 2)
    qc.cx(2, 3)
    qc.rz(np.pi/4, 1)
    qc.rz(np.pi/4, 2)
    qc.rz(np.pi/4, 3)
    angle_w = 2 * np.arcsin(np.sqrt(1/3))
    qc.cry(angle_w, 0, 1)
    qc.cry(angle_w/2, 0, 2)
    
    print(f"  Circuit depth: {qc.depth()}")
    print(f"  Number of gates: {len(qc.data)}")
    print(f"  angle_w = {np.degrees(angle_w):.2f}°")
    
    # Get statevector
    print("\n[1.2] Compute exact statevector")
    sv = Statevector(qc)
    
    # Check normalization
    norm = np.abs(np.vdot(sv.data, sv.data))
    print(f"  Normalization: {norm:.6f} (should be 1.0)")
    
    # Show non-zero amplitudes
    print("\n[1.3] Non-zero amplitudes:")
    amplitudes = {}
    for i, amp in enumerate(sv.data):
        if np.abs(amp) > 1e-10:
            bitstring = format(i, '04b')
            amplitudes[bitstring] = amp
            prob = np.abs(amp)**2
            print(f"    |{bitstring}⟩: amp = {amp.real:+.4f}{amp.imag:+.4f}j, |amp|² = {prob:.4f}")
    
    # Calculate theoretical purity of BCD subsystem
    print("\n[1.4] BCD subsystem purity (theoretical)")
    rho_full = DensityMatrix(sv)
    
    # Trace out qubit A (qubit 0)
    rho_bcd = partial_trace(rho_full, [0])
    
    # Purity = Tr(ρ²)
    purity_theoretical = np.real(np.trace(rho_bcd.data @ rho_bcd.data))
    print(f"  Purity(BCD) = Tr(ρ²) = {purity_theoretical:.4f}")
    print(f"  (1.0 = pure, 0.25 = maximally mixed for 3 qubits)")
    
    if purity_theoretical < 0.9:
        print(f"  ⚠️  BCD is NOT pure! The circuit creates a mixed state.")
    else:
        print(f"  ✓ BCD is pure")
    
    # Calculate theoretical CHSH for qubits 0 and 1
    print("\n[1.5] Theoretical CHSH (qubits A-B)")
    
    # For perfect CHSH measurement, we need:
    # |ψ⟩ on qubits AB should give S = 2√2 for maximally entangled
    
    # Trace out qubits C,D (qubits 2,3)
    rho_ab = partial_trace(rho_full, [2, 3])
    purity_ab = np.real(np.trace(rho_ab.data @ rho_ab.data))
    print(f"  Purity(AB) = {purity_ab:.4f}")
    
    # Calculate actual 2-qubit reduced density matrix
    print(f"  ρ_AB =")
    for i in range(4):
        row = "    "
        for j in range(4):
            val = rho_ab.data[i,j]
            row += f"{val.real:+.3f}{val.imag:+.3f}j  "
        print(row)
    
    # Check if AB is entangled
    from qiskit.quantum_info import concurrence
    try:
        conc = concurrence(rho_ab)
        print(f"\n  Concurrence(AB) = {conc:.4f}")
        print(f"  (0 = separable, 1 = maximally entangled)")
    except:
        print(f"\n  Could not compute concurrence")
    
    return qc, sv, amplitudes, purity_theoretical, purity_ab

def investigate_measurements(qc_base, sv_ideal):
    """Check if measurement bases are applied correctly"""
    
    print("\n" + "="*80)
    print("LEVEL 2: MEASUREMENT VERIFICATION")
    print("="*80)
    
    # Test each Pauli basis
    bases_to_test = ['Z', 'X', 'Y']
    
    simulator = AerSimulator()
    
    for basis in bases_to_test:
        print(f"\n[2.{ord(basis)-87}] Testing {basis} basis measurement on qubit 0")
        
        qc = qc_base.copy()
        
        # Apply basis rotation
        if basis == 'X':
            qc.h(0)
        elif basis == 'Y':
            qc.sdg(0)
            qc.h(0)
        
        qc.measure_all()
        
        # Run measurement
        job = execute(qc, simulator, shots=10000)
        counts = job.result().get_counts()
        
        # Show distribution
        print(f"  Measurement distribution:")
        sorted_counts = sorted(counts.items(), key=lambda x: x[1], reverse=True)
        for bitstring, count in sorted_counts[:6]:
            prob = count / 10000
            print(f"    |{bitstring}⟩: {count:4d} ({prob:.3f})")
        
        # Calculate expectation of Z on qubit 0
        exp_val = 0
        for bitstring, count in counts.items():
            # Qiskit convention: rightmost bit is qubit 0
            bit_0 = int(bitstring[-1])
            sign = 1 if bit_0 == 0 else -1
            exp_val += sign * count / 10000
        
        print(f"  ⟨{basis}₀⟩ = {exp_val:+.4f}")

def investigate_correlations(qc_base):
    """Verify two-qubit correlation calculations"""
    
    print("\n" + "="*80)
    print("LEVEL 3: CORRELATION CALCULATION VERIFICATION")
    print("="*80)
    
    simulator = AerSimulator()
    
    # Test CHSH correlations
    chsh_bases = [
        ('ZZ', []),
        ('ZX', [1]),
        ('XZ', [0]),
        ('XX', [0, 1]),
    ]
    
    correlations = {}
    
    for label, h_qubits in chsh_bases:
        print(f"\n[3.{len(correlations)+1}] Computing E({label}) for qubits 0,1")
        
        qc = qc_base.copy()
        
        # Apply Hadamards for X measurements
        for q in h_qubits:
            qc.h(q)
        
        qc.measure_all()
        
        # Run
        job = execute(qc, simulator, shots=10000)
        counts = job.result().get_counts()
        
        # Method 1: Direct correlation
        corr_direct = 0
        for bitstring, count in counts.items():
            # Extract bits for qubits 0 and 1
            bits = bitstring[::-1]  # Reverse for standard ordering
            bit_0 = int(bits[0])
            bit_1 = int(bits[1])
            
            # Correlation: +1 if same, -1 if different
            if bit_0 == bit_1:
                corr_direct += count / 10000
            else:
                corr_direct -= count / 10000
        
        correlations[label] = corr_direct
        
        print(f"  Method 1 (direct): E({label}) = {corr_direct:+.4f}")
        
        # Method 2: Expectation values
        exp_0 = 0
        exp_1 = 0
        for bitstring, count in counts.items():
            bits = bitstring[::-1]
            
            bit_0 = int(bits[0])
            bit_1 = int(bits[1])
            
            sign_0 = 1 if bit_0 == 0 else -1
            sign_1 = 1 if bit_1 == 0 else -1
            
            exp_0 += sign_0 * count / 10000
            exp_1 += sign_1 * count / 10000
        
        print(f"  ⟨A⟩ = {exp_0:+.4f}, ⟨B⟩ = {exp_1:+.4f}")
        
        # Show sample counts
        sorted_counts = sorted(counts.items(), key=lambda x: x[1], reverse=True)
        print(f"  Top outcomes:")
        for bitstring, count in sorted_counts[:4]:
            bits = bitstring[::-1]
            match = "✓" if bits[0] == bits[1] else "✗"
            print(f"    |{bitstring}⟩: {count:4d} ({count/10000:.3f}) [{match}]")
    
    # Calculate CHSH
    S = correlations['ZZ'] - correlations['ZX'] + correlations['XZ'] + correlations['XX']
    
    print(f"\n  CHSH Parameter:")
    print(f"    S = {correlations['ZZ']:+.4f} - {correlations['ZX']:+.4f} + {correlations['XZ']:+.4f} + {correlations['XX']:+.4f}")
    print(f"    S = {S:.4f}")
    print(f"\n  Classical bound: |S| ≤ 2.000")
    
    if abs(S) > 2.0:
        print(f"    ✓ VIOLATION!")
    else:
        print(f"    ✗ No violation ({abs(S)/2*100:.1f}% of bound)")
    
    return correlations, S

def investigate_purity(qc_base):
    """Debug the purity calculation method"""
    
    print("\n" + "="*80)
    print("LEVEL 4: PURITY CALCULATION VERIFICATION")
    print("="*80)
    
    simulator = AerSimulator()
    
    # Your purity calculation: avg_correlation^2
    print("\n[4.1] Your method: purity ≈ (avg |⟨Pauli⟩|)²")
    
    # Measure several Pauli bases
    pauli_bases = ['ZZZZ', 'XXXX', 'YYYY', 'XZZZ', 'YZZZ', 'XXZZ']
    
    expectations = []
    
    for basis in pauli_bases:
        qc = qc_base.copy()
        
        # Apply basis rotations
        for i, p in enumerate(basis):
            if p == 'X':
                qc.h(i)
            elif p == 'Y':
                qc.sdg(i)
                qc.h(i)
        
        qc.measure_all()
        
        job = execute(qc, simulator, shots=10000)
        counts = job.result().get_counts()
        
        # Calculate expectation
        exp_val = 0
        for bitstring, count in counts.items():
            parity = sum(int(b) for b in bitstring) % 2
            sign = 1 if parity == 0 else -1
            exp_val += sign * count / 10000
        
        expectations.append(abs(exp_val))
        print(f"  |⟨{basis}⟩| = {abs(exp_val):.4f}")
    
    avg_corr = np.mean(expectations)
    purity_method1 = avg_corr ** 2
    
    print(f"\n  Average: {avg_corr:.4f}")
    print(f"  Purity (method 1): {purity_method1:.4f}")
    
    # Correct purity calculation
    print("\n[4.2] Correct method: Tr(ρ²) from density matrix")
    
    sv = Statevector(qc_base)
    rho_full = DensityMatrix(sv)
    rho_bcd = partial_trace(rho_full, [0])
    
    purity_correct = np.real(np.trace(rho_bcd.data @ rho_bcd.data))
    
    print(f"  Purity (correct): {purity_correct:.4f}")
    
    print(f"\n  Discrepancy: {abs(purity_method1 - purity_correct):.4f}")
    
    if abs(purity_method1 - purity_correct) > 0.1:
        print(f"  ⚠️  METHOD 1 IS WRONG!")
        print(f"  The formula (avg |⟨Pauli⟩|)² does NOT give purity")
        print(f"  This explains the low purity reading!")
    
    return purity_method1, purity_correct

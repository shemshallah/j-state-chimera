# TARGET STATE
# ═══════════════════════════════════════════════════════════════════════════

def get_target_state():
    """
    Target Janus state:
    |Ψ⟩ = (|0⟩⊗|GHZ⟩ + |1⟩⊗|W⟩) / normalization
    
    Expanding:
    |0⟩⊗|GHZ⟩ = |0⟩⊗(|000⟩+|111⟩)/√2 = (|0000⟩ + |0111⟩)/√2
    |1⟩⊗|W⟩ = |1⟩⊗(|001⟩+|010⟩+|100⟩)/√3 = (|1001⟩ + |1010⟩ + |1100⟩)/√3
    
    So:
    |Ψ⟩ = [(|0000⟩ + |0111⟩)/√2 + (|1001⟩ + |1010⟩ + |1100⟩)/√3] / √2
    
    Let's just make it an equal superposition for simplicity:
    |Ψ⟩ = (|0000⟩ + |0111⟩ + |1001⟩ + |1010⟩ + |1100⟩) / √5
    """
    state = np.zeros(16, dtype=complex)
    
    # Equal amplitudes (simpler than proper GHZ/W weighting)
    state[0b0000] = 1  # |0000⟩
    state[0b0111] = 1  # |0111⟩
    state[0b1001] = 1  # |1001⟩
    state[0b1010] = 1  # |1010⟩
    state[0b1100] = 1  # |1100⟩
    
    # Normalize
    norm = np.sqrt(np.sum(np.abs(state)**2))
    return state / norm

# ═══════════════════════════════════════════════════════════════════════════
# CIRCUIT USING INITIALIZE
# ═══════════════════════════════════════════════════════════════════════════

def create_janus_with_initialize():
    """
    Use qiskit's Initialize to create exact state
    This CANNOT fail - it's mathematically guaranteed!
    """
    target = get_target_state()
    
    # Create circuit
    qc = QuantumCircuit(4, 4)
    
    # Use Initialize to set exact state
    init = Initialize(target)
    qc.append(init, [0, 1, 2, 3])
    
    # Decompose Initialize into basic gates
    qc = qc.decompose()
    
    return qc

# ═══════════════════════════════════════════════════════════════════════════

def measure_angle(theta_deg):
    """Measure at angle theta"""
    qc = create_janus_with_initialize()
    qc.ry(-np.radians(theta_deg), 0)
    qc.measure([0,1,2,3], [0,1,2,3])
    
    qc_trans = transpile(qc, backend=backend, optimization_level=3)
    
    start = time.time()
    job = backend.run(qc_trans, shots=NUM_SHOTS)
    counts = job.result().get_counts()
    elapsed = time.time() - start
    
    # Extract BCD
    bcd_counts = defaultdict(int)
    for bitstring, count in counts.items():
        bits = bitstring[::-1]
        bcd = bits[1:4]
        bcd_counts[bcd] += count
    
    total = sum(counts.values())
    
    # Probabilities
    probs = {}
    for i in range(8):
        bcd_str = format(i, '03b')
        probs[bcd_str] = bcd_counts.get(bcd_str, 0) / total
    
    # Entropy
    p_list = [p for p in probs.values() if p > 0]
    H = entropy(p_list, base=2)
    
    # Tangle signature
    ghz = probs['000'] + probs['111']
    w = probs['001'] + probs['010'] + probs['100']
    tau = ghz - w
    
    return H, tau, probs, elapsed

    # Analysis
    print("\n\n🔬 PHASE 3: ANALYSIS\n" + "─"*80)
    
    entropies = np.array([r['entropy'] for r in results])
    tangles = np.array([r['tangle'] for r in results])

        print("  ✗ Weak variation")
    
    # Zero crossing

        print(f"\n✓ Zero-crossing detected")
    else:
        print(f"\n✗ No zero-crossing")
    
    # Contextuality

        if C > 0.1:
            print("  ✓ Significant")
        else:
            print("  ✗ Weak")
    
    print("\n" + "═"*80)
    print("COMPLETE")
    print("═"*80)

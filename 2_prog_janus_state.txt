def create_janus_state():
    """Create the Janus state (from v1.0 - this circuit WORKS)"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    
    qc.h(0)
    qc.ch(0, 1)
    qc.cx(0, 2)
    qc.cx(1, 2)
    qc.cx(2, 3)
    qc.rz(np.pi/4, 1)
    qc.rz(np.pi/4, 2)
    qc.rz(np.pi/4, 3)
    
    angle_w = 2 * np.arcsin(np.sqrt(1/3))
    qc.cry(angle_w, 0, 1)
    qc.cry(angle_w/2, 0, 2)
    
    return qc

def create_ghz_state():
    """Pure GHZ on qubits 1,2,3"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    qc.h(1)
    qc.cx(1, 2)
    qc.cx(2, 3)
    return qc

def create_w_state():
    """Pure W on qubits 1,2,3"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    angle_w = 2 * np.arccos(np.sqrt(2/3))
    qc.ry(angle_w, 1)
    qc.ch(1, 2)
    qc.x(1)
    qc.ccx(1, 2, 3)
    qc.x(1)
    return qc

def measure_computational(qc):
    """Measure all in computational basis"""
    qc.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc

def measure_hadamard_A(qc):
    """Measure A in Hadamard, rest computational"""
    qc.h(0)
    qc.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc

def analyze_counts(counts):
    """Extract BCD entropy and 3-tangle estimate"""
    total = sum(counts.values())
    
    # BCD subsystem (ignore A)
    bcd_counts = defaultdict(int)
    for bitstring, count in counts.items():
        bcd = bitstring[-3:]
        bcd_counts[bcd] += count
    
    # Entropy
    probs = np.array([bcd_counts[format(i, '03b')] / total for i in range(8)])
    probs = probs[probs > 0]
    ent = entropy(probs, base=2)
    
    # 3-tangle estimate
    ghz_sig = bcd_counts['000'] + bcd_counts['111']
    w_sig = bcd_counts['001'] + bcd_counts['010'] + bcd_counts['100']
    tangle = (ghz_sig - w_sig) / total
    
    return ent, tangle

def run_trial(circuit_fn, basis_fn, state_name, basis_name, trial_num):
    """Run single trial and return metrics"""
    qc = circuit_fn()
    qc = basis_fn(qc)
    
    start = time.time()
    qc_trans = transpile(qc, backend=backend, optimization_level=2)
    job = backend.run(qc_trans, shots=NUM_SHOTS)
    counts = job.result().get_counts()
    elapsed = time.time() - start
    
    ent, tangle = analyze_counts(counts)
    
    csv_writer.writerow([
        f"trial_{state_name}_{basis_name}", state_name, basis_name,
        ent, tangle, trial_num, elapsed
    ])
    csv_file.flush()
    
    return ent, tangle, elapsed

def analyze_results(janus_comp, janus_had, ghz, w):
    """Comprehensive statistical analysis"""
    
    print("\n" + "="*80)
    print("📊 STATISTICAL ANALYSIS")
    print("="*80)
    
    # 1. Basis-Entropy Divergence (BED)
    jc_ent = np.array(janus_comp['entropy'])
    jh_ent = np.array(janus_had['entropy'])
    
    bed = np.mean(jh_ent) - np.mean(jc_ent)
    bed_percent = (bed / np.mean(jc_ent)) * 100
    
    print(f"\n1. BASIS-ENTROPY DIVERGENCE (BED)")
    print(f"   Janus (comp):    {np.mean(jc_ent):.3f} ± {np.std(jc_ent):.3f}")
    print(f"   Janus (had):     {np.mean(jh_ent):.3f} ± {np.std(jh_ent):.3f}")
    print(f"   BED = ΔH:        {bed:+.3f} ({bed_percent:+.1f}%)")
    
    # T-test for significance
    t_stat, p_value = ttest_ind(jc_ent, jh_ent)
    print(f"   t-statistic:     {t_stat:.3f}")
    print(f"   p-value:         {p_value:.4f}")
    
    if p_value < 0.05:
        print(f"   ✓ SIGNIFICANT (p < 0.05)")
    else:
        print(f"   ✗ NOT SIGNIFICANT")
    
    # 2. 3-Tangle Sign Flip
    jc_tangle = np.array(janus_comp['tangle'])
    jh_tangle = np.array(janus_had['tangle'])
    
    print(f"\n2. THREE-TANGLE BEHAVIOR")
    print(f"   Janus (comp):    {np.mean(jc_tangle):+.3f} ± {np.std(jc_tangle):.3f}")
    print(f"   Janus (had):     {np.mean(jh_tangle):+.3f} ± {np.std(jh_tangle):.3f}")
    
    sign_flip = np.sign(np.mean(jc_tangle)) != np.sign(np.mean(jh_tangle))
    if sign_flip:
        print(f"   ✓ SIGN FLIP CONFIRMED")
    else:
        print(f"   ✗ NO SIGN FLIP")
    
    # 3. Comparison to baselines
    ghz_comp_ent = np.mean(ghz['comp']['entropy'])
    ghz_had_ent = np.mean(ghz['had']['entropy'])
    w_comp_ent = np.mean(w['comp']['entropy'])
    w_had_ent = np.mean(w['had']['entropy'])
    
    print(f"\n3. COMPARISON TO BASELINES")
    print(f"   GHZ basis-dependence: {abs(ghz_had_ent - ghz_comp_ent):.3f}")
    print(f"   W basis-dependence:   {abs(w_had_ent - w_comp_ent):.3f}")
    print(f"   Janus BED:            {abs(bed):.3f}")
    
    if abs(bed) > abs(ghz_had_ent - ghz_comp_ent) and abs(bed) > abs(w_had_ent - w_comp_ent):
        print(f"   ✓ JANUS SHOWS STRONGEST BASIS-DEPENDENCE")
    
    # 4. Publication readiness
    print(f"\n" + "="*80)
    print("📄 PUBLICATION VERDICT")
    print("="*80)
    
    criteria = {
        'bed_significant': (p_value < 0.05, f"BED statistically significant: {p_value < 0.05}"),
        'bed_large': (abs(bed) > 0.3, f"BED > 0.3: {abs(bed):.3f}"),
        'sign_flip': (sign_flip, f"3-tangle sign flip: {sign_flip}"),
        'distinct': (abs(bed) > 0.2, f"Distinct from GHZ/W: {abs(bed) > 0.2}")
    }
    
    passed = sum([v[0] for v in criteria.values()])
    total = len(criteria)
    
    print(f"\nCriteria passed: {passed}/{total}")
    for name, (result, desc) in criteria.items():
        symbol = "✓" if result else "✗"
        print(f"  {symbol} {desc}")
    
    if passed >= 3:
        print(f"\n✅ PUBLICATION READY!")
        print(f"   Evidence supports basis-dependent entanglement")
    else:
        print(f"\n⚠️  MORE DATA NEEDED")
        print(f"   Consider running more trials or optimizing circuit")
    
    return {
        'bed': bed,
        'p_value': p_value,
        'sign_flip': sign_flip,
        'passed_criteria': passed,
        'total_criteria': total
    }

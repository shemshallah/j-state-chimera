def create_entangled_state():
    """Create maximally entangled 4-qubit state"""
    qc = QuantumCircuit(4, 4)
    
    # Bell pairs
    qc.h(0)
    qc.cx(0, 1)
    qc.h(2)
    qc.cx(2, 3)
    
    # Cross-entangle
    qc.cz(1, 2)
    
    return qc

def steering_circuit(theta_deg):
    """Steering protocol circuit"""
    qc = create_entangled_state()
    
    # Rotate qubit 0
    qc.ry(-np.radians(theta_deg), 0)
    
    # Adaptive operations
    qc.cx(0, 1)
    qc.ccx(0, 1, 2)
    
    # W-character when A=1
    qc.x(0)
    qc.cry(np.pi/3, 0, 1)
    qc.cry(np.pi/4, 0, 2)
    qc.x(0)
    
    return qc

def discord_circuit(theta_deg):
    """Discord measurement circuit"""
    qc = create_entangled_state()
    qc.ry(-np.radians(theta_deg), 0)
    qc.cz(0, 2)
    qc.cz(1, 3)
    return qc

def measure_steering_enhanced(theta_deg, shots=NUM_SHOTS):
    """Enhanced steering measurement with full statistics"""
    qc = steering_circuit(theta_deg)
    qc.measure([0,1,2,3], [0,1,2,3])
    
    qc_trans = transpile(qc, backend=backend, optimization_level=3)
    
    start = time.time()
    job = backend.run(qc_trans, shots=shots)
    counts = job.result().get_counts()
    elapsed = time.time() - start
    
    # Separate by qubit A
    counts_A0 = defaultdict(int)
    counts_A1 = defaultdict(int)
    
    for bitstring, count in counts.items():
        bits = bitstring[::-1]
        if bits[0] == '0':
            counts_A0[bits[1:4]] += count
        else:
            counts_A1[bits[1:4]] += count
    
    total_A0 = sum(counts_A0.values())
    total_A1 = sum(counts_A1.values())
    
    # Probabilities
    p_A0 = total_A0 / shots
    p_A1 = total_A1 / shots
    
    # Entropies
    def calc_entropy(counts_dict, total):
        if total == 0:
            return 0
        probs = [c/total for c in counts_dict.values() if c > 0]
        return entropy(probs, base=2) if probs else 0
    
    H_A0 = calc_entropy(counts_A0, total_A0)
    H_A1 = calc_entropy(counts_A1, total_A1)
    
    # Total BCD entropy
    all_bcd = defaultdict(int)
    for bitstring, count in counts.items():
        bits = bitstring[::-1]
        all_bcd[bits[1:4]] += count
    H_total = calc_entropy(all_bcd, shots)
    
    # Steering parameter
    steering = H_total - (p_A0 * H_A0 + p_A1 * H_A1)
    
    # Entanglement witnesses
    # GHZ witness: P(000) + P(111)
    ghz_A0 = (counts_A0.get('000', 0) + counts_A0.get('111', 0)) / max(total_A0, 1)
    ghz_A1 = (counts_A1.get('000', 0) + counts_A1.get('111', 0)) / max(total_A1, 1)
    ghz_witness = (p_A0 * ghz_A0 + p_A1 * ghz_A1)
    
    # W witness: P(001) + P(010) + P(100)
    w_A0 = (counts_A0.get('001', 0) + counts_A0.get('010', 0) + counts_A0.get('100', 0)) / max(total_A0, 1)
    w_A1 = (counts_A1.get('001', 0) + counts_A1.get('010', 0) + counts_A1.get('100', 0)) / max(total_A1, 1)
    w_witness = (p_A0 * w_A0 + p_A1 * w_A1)
    
    # Three-tangle estimate
    tau_est = ghz_witness - w_witness
    
    return {
        'steering': steering,
        'H_total': H_total,
        'H_A0': H_A0,
        'H_A1': H_A1,
        'entropy_diff': abs(H_A0 - H_A1),
        'ghz_witness': ghz_witness,
        'w_witness': w_witness,
        'tau_est': tau_est,
        'p_A0': p_A0,
        'p_A1': p_A1,
        'time': elapsed
    }

def measure_discord_enhanced(theta_deg, shots=NUM_SHOTS):
    """Enhanced discord measurement"""
    qc = discord_circuit(theta_deg)
    qc.measure([0,1,2,3], [0,1,2,3])
    
    qc_trans = transpile(qc, backend=backend, optimization_level=3)
    job = backend.run(qc_trans, shots=shots)
    counts = job.result().get_counts()
    
    total = sum(counts.values())
    
    # Marginals
    p_A = defaultdict(int)
    p_BCD = defaultdict(int)
    
    for bitstring, count in counts.items():
        bits = bitstring[::-1]
        p_A[bits[0]] += count
        p_BCD[bits[1:4]] += count
    
    H_A = entropy([c/total for c in p_A.values()], base=2)
    H_BCD = entropy([c/total for c in p_BCD.values() if c > 0], base=2)
    
    # Joint
    p_full = [c/total for c in counts.values()]
    H_full = entropy(p_full, base=2)
    
    # Mutual information
    I_A_BCD = H_A + H_BCD - H_full
    
    # Discord (simplified)
    discord = max(0, I_A_BCD)
    
    return {
        'discord': discord,
        'mutual_info': I_A_BCD
    }

def steering_model(theta, a, b):
    """Model: S(θ) = a·sin²(θ) + b"""
    return a * np.sin(np.radians(theta))**2 + b

def discord_model(theta, c, d):
    """Model: D(θ) = c·cos²(θ) + d"""
    return c * np.cos(np.radians(theta))**2 + d

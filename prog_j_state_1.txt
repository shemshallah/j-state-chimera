def create_ghz_3():
    """Create 3-qubit GHZ state: (|000⟩ + |111⟩)/√2"""
    qc = QuantumCircuit(3)
    qc.h(0)
    qc.cx(0, 1)
    qc.cx(1, 2)
    return qc

def create_w_3():
    """Create 3-qubit W state: (|001⟩ + |010⟩ + |100⟩)/√3"""
    qc = QuantumCircuit(3)
    # W state construction
    qc.ry(2 * np.arccos(np.sqrt(2/3)), 0)
    qc.ch(0, 1)
    qc.x(0)
    qc.ccx(0, 1, 2)
    qc.x(0)
    return qc

def create_conditional_entanglement_state():
    """
    Create the conditional entanglement state:
    |Ψ_C⟩ = 1/√2(|+⟩_A ⊗ |GHZ⟩_BCD + |-⟩_A ⊗ |W⟩_BCD)
    
    Strategy: Use controlled operations where A controls GHZ vs W
    """
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    
    # Put A in |+⟩ state
    qc.h(0)
    
    # Controlled-GHZ: when A=|0⟩, create GHZ on BCD
    # H on B controlled by A=0
    qc.x(0)  # Flip so control is on |0⟩
    qc.ch(0, 1)  # Controlled-H
    qc.x(0)  # Flip back
    
    # CX gates for GHZ (controlled by A=0)
    qc.x(0)
    qc.ccx(0, 1, 2)
    qc.ccx(0, 2, 3)
    qc.x(0)
    
    # Controlled-W: when A=|1⟩, create W on BCD
    # This is trickier - use controlled rotations
    angle_w = 2 * np.arccos(np.sqrt(2/3))
    qc.cry(angle_w, 0, 1)
    
    # Controlled operations for W state
    qc.ch(0, 2)
    qc.ccx(0, 1, 2)
    qc.cx(0, 3)
    qc.ccx(0, 2, 3)
    
    return qc

def measure_computational_basis(qc):
    """Measure all qubits in computational basis"""
    qc.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc

def measure_hadamard_basis_A(qc):
    """Measure A in Hadamard basis, others in computational"""
    qc.h(0)  # Transform to |+⟩/|-⟩ basis
    qc.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc

def analyze_measurement_results(counts, test_id, circuit_type, basis):
    """Analyze quantum measurement outcomes"""
    total = sum(counts.values())
    
    # Extract BCD subsystem (ignore A)
    bcd_counts = defaultdict(int)
    for bitstring, count in counts.items():
        # Bitstring is ABCD, we want BCD
        bcd = bitstring[-3:]  # Last 3 bits
        bcd_counts[bcd] += count
    
    # Calculate entropy of BCD subsystem
    probs = np.array([bcd_counts[format(i, '03b')] / total for i in range(8)])
    probs = probs[probs > 0]
    ent = entropy(probs, base=2)
    
    # Calculate average correlation
    correlations = []
    bitstrings = []
    for bs, count in counts.items():
        bitstrings.extend([bs] * count)
    
    if len(bitstrings) > 0:
        bit_array = np.array([[int(b) for b in bs] for bs in bitstrings])
        if bit_array.shape[0] > 1 and bit_array.shape[1] > 1:
            corr_matrix = np.corrcoef(bit_array.T)
            off_diag = corr_matrix[np.triu_indices_from(corr_matrix, k=1)]
            avg_corr = np.mean(np.abs(off_diag)) if len(off_diag) > 0 else 0
        else:
            avg_corr = 0
    else:
        avg_corr = 0
    
    # Estimate 3-tangle (rough approximation)
    # For BCD: check if concentrated on |000⟩ + |111⟩ (GHZ-like, high tangle)
    # or distributed (W-like, low tangle)
    ghz_signature = bcd_counts['000'] + bcd_counts['111']
    w_signature = bcd_counts['001'] + bcd_counts['010'] + bcd_counts['100']
    three_tangle_est = (ghz_signature - w_signature) / total
    
    # Find dominant states
    sorted_states = sorted(counts.items(), key=lambda x: x[1], reverse=True)
    dominant = ', '.join([f"{bs}:{c}" for bs, c in sorted_states[:3]])
    
    return {
        'entropy_BCD': ent,
        'correlation_avg': avg_corr,
        'three_tangle_estimate': three_tangle_est,
        'dominant_states': dominant
    }

def test_1_baseline_ghz_and_w():

    # Put GHZ on last 3 qubits (B, C, D)
    qc_ghz.h(1)
    qc_ghz.cx(1, 2)
    qc_ghz.cx(2, 3)
    qc_ghz.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    
    start = time.time()
    qc_trans = transpile(qc_ghz, backend=backend, optimization_level=2)
    job = backend.run(qc_trans, shots=NUM_SHOTS)
    counts_ghz = job.result().get_counts()
    elapsed = time.time() - start
    
    analysis = analyze_measurement_results(counts_ghz, 1, 'GHZ_baseline', 'computational')
    results.append(('GHZ', analysis, elapsed))
    
    print(f"    ✓ GHZ entropy: {analysis['entropy_BCD']:.3f}")
    print(f"    ✓ 3-tangle est: {analysis['three_tangle_estimate']:.3f}")
    print(f"    ✓ Time: {elapsed:.2f}s")
    
    csv_writer.writerow([1, 'GHZ_baseline', 'computational',
                        analysis['entropy_BCD'], analysis['correlation_avg'],
                        analysis['three_tangle_estimate'], 
                        analysis['dominant_states'], elapsed])
    csv_file.flush()
    
    # Test W
    print("\n  Creating pure W state on qubits 1-3...")

    qc_w.x(1)
    qc_w.ccx(1, 2, 3)

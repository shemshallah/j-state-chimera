def create_janus_state():
    """Janus state circuit"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    qc.h(0)
    qc.ch(0, 1)
    qc.cx(0, 2)
    qc.cx(1, 2)
    qc.cx(2, 3)
    qc.rz(np.pi/4, 1)
    qc.rz(np.pi/4, 2)
    qc.rz(np.pi/4, 3)
    angle_w = 2 * np.arcsin(np.sqrt(1/3))
    qc.cry(angle_w, 0, 1)
    qc.cry(angle_w/2, 0, 2)
    return qc

def measure_at_angle(qc, theta, qubit=0):
    """Measure specified qubit at angle theta in X-Z plane"""
    qc_copy = qc.copy()
    qc_copy.ry(-theta, qubit)
    qc_copy.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc_copy

def measure_pauli_basis(qc, basis_string):
    """
    Measure in Pauli basis specified by string
    basis_string: e.g., 'XYZZ' means X on qubit 0, Y on 1, Z on 2,3
    """
    qc_copy = qc.copy()
    
    for i, basis in enumerate(basis_string):
        if basis == 'X':
            qc_copy.h(i)
        elif basis == 'Y':
            qc_copy.sdg(i)
            qc_copy.h(i)
        # Z basis needs no rotation
    
    qc_copy.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    return qc_copy

def analyze_counts(counts):
    """Extract entropy and 3-tangle"""
    total = sum(counts.values())
    
    # BCD subsystem
    bcd_counts = defaultdict(int)
    for bitstring, count in counts.items():
        bcd = bitstring[-3:]
        bcd_counts[bcd] += count
    
    # Entropy
    probs = np.array([bcd_counts[format(i, '03b')] / total for i in range(8)])
    probs = probs[probs > 0]
    ent = entropy(probs, base=2)
    
    # 3-tangle estimate
    ghz_sig = bcd_counts['000'] + bcd_counts['111']
    w_sig = bcd_counts['001'] + bcd_counts['010'] + bcd_counts['100']
    tangle = (ghz_sig - w_sig) / total
    
    return ent, tangle, counts

def run_measurement(circuit_fn, measure_fn, shots=NUM_SHOTS):
    """Execute and analyze"""
    qc = circuit_fn()
    qc = measure_fn(qc)
    
    start = time.time()
    qc_trans = transpile(qc, backend=backend, optimization_level=2)
    job = backend.run(qc_trans, shots=shots)
    counts = job.result().get_counts()
    elapsed = time.time() - start
    
    ent, tangle, full_counts = analyze_counts(counts)
    return ent, tangle, full_counts, elapsed

def compute_expectation(counts, observable):
    """
    Compute expectation value of Pauli observable
    observable: dict mapping bitstrings to eigenvalues
    """
    total = sum(counts.values())
    expectation = 0
    
    for bitstring, count in counts.items():
        # Get parity
        parity = sum([int(b) for b in bitstring]) % 2
        sign = 1 if parity == 0 else -1
        expectation += sign * count / total
    
    return expectation

def experiment_1_zero_crossing():
    """
    Adaptive search to find exact angle where τ = 0
    
    Strategy:
    1. Start with bounds [45°, 75°] from v3.0 data
    2. Use golden section search
    3. Refine to ±2° precision
    
    Measurements: ~8-10 (adaptive)
    """
    print("\n" + "="*80)
    print("EXPERIMENT 1: ZERO-CROSSING HUNT")
    print("="*80)
    print("Goal: Find exact angle θ* where τ_BCD = 0\n")
    
    # Initial bounds from v3.0 results
    theta_low = np.pi/4      # 45°: τ = +0.144
    theta_high = np.pi/3     # 60°: τ = -0.038
    
    print(f"  Initial bounds: [{np.degrees(theta_low):.1f}°, {np.degrees(theta_high):.1f}°]\n")
    
    measurements = []
    measurement_count = 0
    
    # Golden section search
    golden_ratio = (1 + np.sqrt(5)) / 2
    
    while theta_high - theta_low > np.radians(2):  # 2° precision
        # Choose test points
        theta1 = theta_high - (theta_high - theta_low) / golden_ratio
        theta2 = theta_low + (theta_high - theta_low) / golden_ratio
        
        # Measure at theta1
        print(f"  Test 1: θ = {np.degrees(theta1):.1f}°...", end='', flush=True)
        ent1, tau1, counts1, t1 = run_measurement(
            create_janus_state,
            lambda qc: measure_at_angle(qc, theta1)
        )
        print(f" τ = {tau1:+.3f} ({t1:.1f}s)")
        
        measurements.append((theta1, ent1, tau1))
        measurement_count += 1
        
        csv_writer.writerow([
            'zero_crossing', f'theta_{np.degrees(theta1):.1f}deg',
            ent1, tau1, 0, 0, t1
        ])
        csv_file.flush()
        
        # Measure at theta2
        print(f"  Test 2: θ = {np.degrees(theta2):.1f}°...", end='', flush=True)
        ent2, tau2, counts2, t2 = run_measurement(
            create_janus_state,
            lambda qc: measure_at_angle(qc, theta2)
        )
        print(f" τ = {tau2:+.3f} ({t2:.1f}s)")
        
        measurements.append((theta2, ent2, tau2))
        measurement_count += 1
        
        csv_writer.writerow([
            'zero_crossing', f'theta_{np.degrees(theta2):.1f}deg',
            ent2, tau2, 0, 0, t2
        ])
        csv_file.flush()
        
        # Update bounds based on signs
        if np.sign(tau1) != np.sign(tau2):
            # Zero is between theta1 and theta2
            theta_low = theta1
            theta_high = theta2
            print(f"  → Narrowed to [{np.degrees(theta_low):.1f}°, {np.degrees(theta_high):.1f}°]\n")
        elif abs(tau1) < abs(tau2):
            # theta1 is closer to zero
            theta_high = theta2
            print(f"  → Shifted lower: [{np.degrees(theta_low):.1f}°, {np.degrees(theta_high):.1f}°]\n")
        else:
            # theta2 is closer to zero
            theta_low = theta1
            print(f"  → Shifted higher: [{np.degrees(theta_low):.1f}°, {np.degrees(theta_high):.1f}°]\n")
    
    # Final measurement at midpoint
    theta_star = (theta_low + theta_high) / 2
    print(f"  Final refinement: θ* = {np.degrees(theta_star):.1f}°...", end='', flush=True)
    ent_star, tau_star, counts_star, t_star = run_measurement(
        create_janus_state,
        lambda qc: measure_at_angle(qc, theta_star)
    )
    print(f" τ = {tau_star:+.4f} ({t_star:.1f}s)")
    
    measurements.append((theta_star, ent_star, tau_star))
    measurement_count += 1
    
    csv_writer.writerow([
        'zero_crossing', f'theta_star_{np.degrees(theta_star):.1f}deg',
        ent_star, tau_star, 0, 0, t_star
    ])
    csv_file.flush()
    
    # Results
    print(f"\n  🎯 ZERO-CROSSING RESULTS:")
    print(f"    θ* = {np.degrees(theta_star):.2f}° ± {np.degrees(theta_high - theta_low)/2:.2f}°")
    print(f"    τ(θ*) = {tau_star:+.4f}")
    print(f"    H(θ*) = {ent_star:.3f} bits")
    print(f"    \n    Physical interpretation:")
    print(f"    At {np.degrees(theta_star):.1f}°, BCD is EXACTLY at the transition")
    print(f"    between GHZ-like and W-like correlations!")
    
    return measurements, measurement_count

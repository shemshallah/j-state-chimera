def create_janus_state():
    """Janus state: coherent superposition of GHZ and W topologies"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    
    # Control qubit superposition
    qc.h(0)
    
    # Entanglement generation
    qc.ch(0, 1)
    qc.cx(0, 2)
    qc.cx(1, 2)
    qc.cx(2, 3)
    
    # Phase structure
    qc.rz(np.pi/4, 1)
    qc.rz(np.pi/4, 2)
    qc.rz(np.pi/4, 3)
    
    # W-component
    angle_w = 2 * np.arcsin(np.sqrt(1/3))
    qc.cry(angle_w, 0, 1)
    qc.cry(angle_w/2, 0, 2)
    
    return qc

def measure_at_angle(qc, theta):
    """
    Measure qubit 0 at angle theta in X-Z plane
    theta=0: Z-basis (computational)
    theta=π/2: X-basis (Hadamard)
    """
    qc_copy = qc.copy()
    
    # Rotate measurement basis for qubit 0
    qc_copy.ry(-theta, 0)
    
    # Measure all qubits
    qc_copy.measure(range(NUM_QUBITS), range(NUM_QUBITS))
    
    return qc_copy

def create_ghz_state():
    """Pure GHZ on qubits 1,2,3"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    qc.h(1)
    qc.cx(1, 2)
    qc.cx(2, 3)
    return qc

def create_w_state():
    """Pure W on qubits 1,2,3"""
    qc = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
    angle_w = 2 * np.arccos(np.sqrt(2/3))
    qc.ry(angle_w, 1)
    qc.ch(1, 2)
    qc.x(1)
    qc.ccx(1, 2, 3)
    qc.x(1)
    return qc

def analyze_counts(counts):
    """Extract all key metrics from measurement results - ROBUST VERSION"""
    total = sum(counts.values())
    
    # BCD subsystem (trace out A)
    bcd_counts = defaultdict(int)
    for bitstring, count in counts.items():
        bcd = bitstring[-3:]
        bcd_counts[bcd] += count
    
    # Entropy of BCD
    probs = np.array([bcd_counts[format(i, '03b')] / total for i in range(8)])
    probs = probs[probs > 0]
    ent = entropy(probs, base=2)
    
    # 3-tangle estimate
    ghz_sig = bcd_counts['000'] + bcd_counts['111']
    w_sig = bcd_counts['001'] + bcd_counts['010'] + bcd_counts['100']
    tangle = (ghz_sig - w_sig) / total
    
    # Global correlations - FIXED to handle edge cases
    try:
        bitstrings = []
        for bs, count in counts.items():
            bitstrings.extend([bs] * min(count, 50))  # Limit to 50 per outcome
        
        if len(bitstrings) > 10:
            bit_array = np.array([[int(b) for b in bs] for bs in bitstrings[:500]])
            
            # Check for variance before computing correlation
            if bit_array.shape[0] > 1 and bit_array.shape[1] > 1:
                # Compute standard deviations
                stds = np.std(bit_array, axis=0)
                
                # Only compute correlation if there's variance
                if np.all(stds > 1e-10):
                    corr_matrix = np.corrcoef(bit_array.T)
                    # Check for NaN
                    if not np.any(np.isnan(corr_matrix)):
                        avg_corr = np.mean(np.abs(corr_matrix[np.triu_indices_from(corr_matrix, k=1)]))
                    else:
                        avg_corr = 0.0
                else:
                    avg_corr = 0.0
            else:
                avg_corr = 0.0
        else:
            avg_corr = 0.0
    except Exception as e:
        print(f"      Warning: Correlation calculation failed: {e}")
        avg_corr = 0.0
    
    return {
        'entropy': ent,
        'tangle': tangle,
        'correlation': avg_corr,
        'bcd_counts': bcd_counts,
        'full_counts': counts
    }

def run_measurement(circuit_fn, measure_fn, shots=NUM_SHOTS):
    """Execute quantum circuit and return results"""
    try:
        qc = circuit_fn()
        qc = measure_fn(qc)
        
        start = time.time()
        qc_trans = transpile(qc, backend=backend, optimization_level=2)
        job = backend.run(qc_trans, shots=shots)
        counts = job.result().get_counts()
        elapsed = time.time() - start
        
        return analyze_counts(counts), elapsed
    except Exception as e:
        print(f"      ERROR in measurement: {e}")
        raise

def test_2_basis_sweep():
    """
    Sweep measurement angle from 0 to π/2
    Measurements: 12 (6 angles × 2 trials)
    """
    print("\n" + "="*80)
    print("TEST 2: BASIS-SWEEP CHARACTERIZATION")
    print("="*80)
    print("Goal: Map H_BCD(θ) to show continuous basis-dependence\n")
    
    angles = [0, np.pi/6, np.pi/4, np.pi/3, np.pi/2.2, np.pi/2]
    angle_names = ['0°', '30°', '45°', '60°', '82°', '90°']
    
    results = {angle: [] for angle in angles}
    
    print(f"  Sweeping through {len(angles)} angles (2 trials each)...\n")
    
    for i, (theta, name) in enumerate(zip(angles, angle_names)):
        print(f"  [{i+1}/{len(angles)}] θ = {name}:")
        
        for trial in range(2):
            print(f"    Trial {trial+1}/2...", end='', flush=True)
            result, t = run_measurement(
                create_janus_state,
                lambda qc, th=theta: measure_at_angle(qc, th)
            )
            results[theta].append(result)
            
            print(f" H={result['entropy']:.3f}, τ={result['tangle']:+.3f} ({t:.1f}s)")
            
            csv_writer.writerow([
                'basis_sweep', f'theta_{name}_trial{trial}',
                result['entropy'], result['tangle'], theta, result['correlation'], t
            ])
            csv_file.flush()
    
    # Analyze trend
    print(f"\n  📊 BASIS-SWEEP RESULTS:")
    avg_entropies = [np.mean([r['entropy'] for r in results[theta]]) for theta in angles]
    
    print(f"    H(0°)  = {avg_entropies[0]:.3f}")
    print(f"    H(90°) = {avg_entropies[-1]:.3f}")
    print(f"    ΔH = {avg_entropies[-1] - avg_entropies[0]:+.3f} ({100*(avg_entropies[-1]/avg_entropies[0]-1):+.1f}%)")
    
    try:
        def model(theta, a, b):
            return a + b * np.sin(theta)**2
        
        popt, _ = curve_fit(model, angles, avg_entropies)
        print(f"    Fit: H(θ) = {popt[0]:.3f} + {popt[1]:.3f}·sin²(θ)")
        print(f"    ✓ CONTINUOUS BASIS-DEPENDENCE CONFIRMED")
    except Exception as e:
        print(f"    ✓ CLEAR TREND OBSERVED (fit failed: {e})")
    
    return results, len(angles) * 2

        print(f"    Hadamard (3 trials):")

    else:
